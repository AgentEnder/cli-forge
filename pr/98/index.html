CLI FORGE

A type-safe CLI builder for Node.js with first-class TypeScript support

GET STARTEDVIEW EXAMPLES
hello-world.ts
import { cli } from 'cli-forge';

cli('hello')
  .option('name', { type: 'string' })
  .command('greet', {
    handler: (args) => {
      // args.name is fully typed
      console.log(`Hello, ${args.name}!`)
    }
  })
  .forge();
